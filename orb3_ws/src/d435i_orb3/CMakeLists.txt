cmake_minimum_required(VERSION 3.16)
project(d435i_orb3)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(message_filters REQUIRED)
find_package(OpenCV 4.4 REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Pangolin REQUIRED)
set(ORB_SLAM3_ROOT "$ENV{HOME}/orb3_ws/vendor/ORB_SLAM3" CACHE PATH "ORB-SLAM3 source root")
if(NOT EXISTS "${ORB_SLAM3_ROOT}/lib/libORB_SLAM3.so")
  message(FATAL_ERROR "Build ORB-SLAM3 first: run tools/deploy.sh")
endif()
add_library(d435i_orb3_backend SHARED src/backend.cpp)
# Compile all Eigen/Sophus exchanges with the same standard and SIMD ABI as ORB.
set_target_properties(d435i_orb3_backend PROPERTIES CXX_STANDARD 14)
target_compile_options(d435i_orb3_backend PRIVATE -march=native)
target_include_directories(d435i_orb3_backend PRIVATE
  "${ORB_SLAM3_ROOT}"
  "${ORB_SLAM3_ROOT}/include" "${ORB_SLAM3_ROOT}/include/CameraModels"
  "${ORB_SLAM3_ROOT}/Thirdparty/Sophus" ${Pangolin_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(d435i_orb3_backend PRIVATE "${ORB_SLAM3_ROOT}/lib/libORB_SLAM3.so"
  Eigen3::Eigen ${OpenCV_LIBS} ${Pangolin_LIBRARIES})
add_executable(orb3_node src/orb3_node.cpp)
target_include_directories(orb3_node PRIVATE ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(orb3_node rclcpp sensor_msgs geometry_msgs nav_msgs std_msgs std_srvs cv_bridge message_filters)
target_link_libraries(orb3_node d435i_orb3_backend ${OpenCV_LIBS})
set_target_properties(orb3_node d435i_orb3_backend PROPERTIES INSTALL_RPATH
  "${CMAKE_INSTALL_PREFIX}/lib;${ORB_SLAM3_ROOT}/lib;${ORB_SLAM3_ROOT}/Thirdparty/DBoW2/lib;${ORB_SLAM3_ROOT}/Thirdparty/g2o/lib;${Pangolin_LIBRARY_DIRS};$ENV{HOME}/orb3_ws/deps/lib")
install(TARGETS d435i_orb3_backend LIBRARY DESTINATION lib)
install(TARGETS orb3_node DESTINATION lib/${PROJECT_NAME})
ament_package()
